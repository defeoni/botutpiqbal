        function renderDecisionBadge(dec) {
            if (!dec.decided) return `<span style="color:var(--text-muted);">Pending</span>`;
            if (dec.choice === 'Disetujui') return `<strong style="color:var(--teal-primary);">Disetujui (${dec.timestamp})</strong>`;
            if (dec.choice === 'Ditolak') return `<strong style="color:var(--red-alert);">Ditolak (${dec.timestamp})</strong>`;
            return `<strong style="color:var(--yellow-alert);">${dec.choice} (${dec.timestamp})</strong>`;
        }

        function checkApprovalFulfillment(item) {
            if (item.klasifikasi === 'By Owner' || item.klasifikasi === 'By Other') return true;
            if (item.jalurRequired === 'PM') return item.decisions.PM.decided && item.decisions.PM.choice === 'Disetujui';
            if (item.jalurRequired === 'MK') return item.decisions.MK.decided && item.decisions.MK.choice === 'Disetujui';
            if (item.jalurRequired === 'HO') return item.decisions.HO.decided && item.decisions.HO.choice === 'Disetujui';
            return false;
        }

                hoCoveredItemIds = hoCoveredItemIds.filter(id => id !== itemId);
            }

            const klas = document.getElementById('temp-klasifikasi').value;
            const merek = document.getElementById('temp-merek').value.trim();

            if (!nama) { alert("Nama item sementara wajib diisi."); return; }

            sessionTempCounter++;
            const newId = "TEMP-" + sessionTempCounter;
            const newItem = {
                id: newId,
                nama: nama,
                klasifikasi: klas,
                merek: merek || "Belum diketahui",
                isMaster: false, // Item Sementara flag
                refApproval: "Item Sementara Non-Master"
            };

            masterKatalog.unshift(newItem);
            renderKatalog(masterKatalog);
            addToDraft(newId);
            closeModal('modal-sementara');
        }

        function prosesTerbitkanRevisi() {
            const pihak = document.getElementById('revisi-pihak').value;
            const alasan = document.getElementById('revisi-alasan').value.trim();

            if (!alasan) { alert("Alasan resmi penerbitan revisi wajib diisi."); return; }

            revVersion++;
            document.getElementById('rev-badge').innerText = `Dokumen: REV.${revVersion}`;
            showToast(`Revisi Resmi REV.${revVersion} Diterbitkan oleh ${pihak}.`);
            closeModal('modal-revisi');
        }

        function prosesBatalItem() {
            const index = document.getElementById('batal-item-index').value;
            const alasan = document.getElementById('batal-item-alasan').value.trim();

            if (!alasan) { alert("Alasan pembatalan item wajib diisi."); return; }

            draftItems[index].statusReview = "Batal";
            draftItems[index].cancelledReason = alasan;
            addHistory(index, "Status Item", "Aktif", "Batal", "Admin", `Dibatalkan: ${alasan}`);

            renderReviewTable();
            closeModal('modal-batal-item');
            showToast("Item berhasil dibatalkan dengan riwayat tersimpan.");
        }

        function getFormattedTime() {
            const now = new Date();
            return now.getHours().toString().padStart(2, '0') + ":" + 
                   now.getMinutes().toString().padStart(2, '0');
        }

        function showToast(msg) {
            const toast = document.getElementById('toast');
            toast.innerText = msg;
            toast.style.display = "block";
            setTimeout(() => { toast.style.display = "none"; }, 3500);
        }
